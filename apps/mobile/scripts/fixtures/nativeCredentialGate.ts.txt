import type { User } from '@/api/contracts';
import { encodeCredential } from './sessionCredential';

/** This adapter exists only in the isolated acceptance checkout. No credential reaches the gate. */
export async function nativeCredentialValue(environment: string, token: string, user?: User) {
  if (
    !__DEV__ ||
    environment !== process.env.EXPO_PUBLIC_API_BASE_URL ||
    user?.id !== process.env.EXPO_PUBLIC_NATIVE_ACCOUNT
  )
    return encodeCredential(token, user);
  const url = process.env.EXPO_PUBLIC_NATIVE_SQL_GATE_URL;
  if (!url) throw new Error('ISOLATED_NATIVE_GATE_REQUIRED');
  const event = (stage: string) =>
    fetch(`${url}/event?stage=${stage}&id=00000000-0000-4000-8000-000000000001`);
  if (!(await event('credential-pause')).ok) throw new Error('NATIVE_CREDENTIAL_GATE_FAILURE');
  if (!(await event('credential-set')).ok) throw new Error('NATIVE_CREDENTIAL_STORE_FAILURE');
  const legacy = await event('credential-legacy');
  if (!legacy.ok) throw new Error('NATIVE_CREDENTIAL_GATE_FAILURE');
  // The SDK still performs a real SecureStore write; 201 selects only the legacy fixture format.
  return legacy.status === 201 ? token : encodeCredential(token, user);
}
